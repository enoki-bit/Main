INSERT INTO m_社員
(
    登録日時
   ,登録者
   ,登録機能ID
   ,更新日時
   ,更新者
   ,更新機能ID
   ,内部更新日時
   ,内部更新者
   ,内部更新機能ID
   ,削除FLG
   ,社員CD
   ,氏名
   ,氏名カナ
   ,所属部署CD
   ,役職CD
   ,性別区分
   ,電話番号
)
SELECT
    GETDATE() AS 登録日時
   ,'永野浩志' AS 登録者
   ,m_sya.登録機能ID
   ,CURRENT_TIMESTAMP() AS 更新日時
   ,'永野浩志' AS 更新者
   ,m_sya.更新機能ID
   ,NOW() AS 内部更新日時
   ,'永野浩志' AS 内部更新者
   ,m_sya.内部更新機能ID
   ,m_sya.削除FLG
   ,m_sya_max.社員CD最大値 + 1 AS 社員CD
   ,m_sya.氏名
   ,m_sya.氏名カナ
   ,m_sya.所属部署CD
   ,m_sya.役職CD
   ,m_sya.性別区分
   ,m_sya.電話番号
FROM m_社員 m_sya
LEFT JOIN
    (
        SELECT
            MAX(社員CD) AS 社員CD最大値
        FROM m_社員
    ) m_sya_max
    ON 1 = 1
WHERE m_sya.社員CD = 9;
